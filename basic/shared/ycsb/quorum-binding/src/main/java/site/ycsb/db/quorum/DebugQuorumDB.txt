package site.ycsb.db.quorum;

import io.grpc.ManagedChannel;
import io.grpc.ManagedChannelBuilder;
import io.grpc.StatusRuntimeException;

import kv.AgentKVGrpc;
import kv.Kv;

import site.ycsb.*;

import java.util.*;
import java.util.concurrent.TimeUnit;

public class QuorumDB extends DB {

    private ManagedChannel channel;
    private AgentKVGrpc.AgentKVBlockingStub stub;

    private String coordinatorHost;
    private int coordinatorPort;

    private static final long RPC_DEADLINE_SECONDS = 10;

    // Debug counters
    private int insertFailures = 0;
    private int insertExceptions = 0;
    private int readExceptions = 0;
    private int deleteExceptions = 0;

    @Override
    public void init() throws DBException {

        Properties props = getProperties();

        coordinatorHost = props.getProperty("coord.host", "10.0.0.101");
        coordinatorPort = Integer.parseInt(
                props.getProperty("coord.port", "6000")
        );

        channel = ManagedChannelBuilder
                .forAddress(coordinatorHost, coordinatorPort)
                .usePlaintext()
                .build();

        // Create base stub once.
        // Apply deadlines per call, not here.
        stub = AgentKVGrpc.newBlockingStub(channel);
    }

    @Override
    public void cleanup() throws DBException {

        System.out.println("=== DEBUG SUMMARY ===");
        System.out.println("Insert failures: " + insertFailures);
        System.out.println("Insert exceptions: " + insertExceptions);
        System.out.println("Read exceptions: " + readExceptions);
        System.out.println("Delete exceptions: " + deleteExceptions);

        if (channel != null) {
            try {
                channel.shutdown().awaitTermination(5, TimeUnit.SECONDS);
            } catch (InterruptedException e) {
                throw new DBException(e);
            }
        }
    }

    @Override
    public Status read(String table,
                       String key,
                       Set<String> fields,
                       Map<String, ByteIterator> result) {

        Kv.AgentGetRequest req =
                Kv.AgentGetRequest.newBuilder()
                        .setKey(key)
                        .build();

        Kv.AgentGetReply reply;

        try {
            reply = stub
                    .withDeadlineAfter(RPC_DEADLINE_SECONDS, TimeUnit.SECONDS)
                    .get(req);

        } catch (StatusRuntimeException e) {

            readExceptions++;

            System.err.println("[READ GRPC ERROR] key=" + key +
                    " status=" + e.getStatus());

            return Status.ERROR;

        } catch (Exception e) {

            readExceptions++;

            System.err.println("[READ EXCEPTION] key=" + key);
            e.printStackTrace();

            return Status.ERROR;
        }

        switch (reply.getStatus()) {

            case OK:
                result.put("field0", new StringByteIterator(reply.getValue()));
                return Status.OK;

            case NOT_FOUND:
                return Status.NOT_FOUND;

            case QUORUM_FAILED:
                System.err.println("[READ QUORUM FAILED] key=" + key);
                return Status.SERVICE_UNAVAILABLE;

            default:
                System.err.println("[READ UNKNOWN STATUS] key=" + key +
                        " status=" + reply.getStatus());
                return Status.ERROR;
        }
    }

    @Override
    public Status insert(String table,
                         String key,
                         Map<String, ByteIterator> values) {

        ByteIterator val = values.get("field0");

        if (val == null) {
            System.err.println("[INSERT ERROR] Missing field0 key=" + key);

            if (values == null || values.isEmpty()) {
                System.err.println("  values map is empty");
            } else {
                for (Map.Entry<String, ByteIterator> entry : values.entrySet()) {
                    System.err.println("  " + entry.getKey() + " = " + entry.getValue());
                }
            }

            insertFailures++;
            return Status.ERROR;
        }

        Kv.AgentPutRequest req =
                Kv.AgentPutRequest.newBuilder()
                        .setKey(key)
                        .setValue(val.toString())
                        .build();

        Kv.AgentPutReply reply;

        try {
            reply = stub
                    .withDeadlineAfter(RPC_DEADLINE_SECONDS, TimeUnit.SECONDS)
                    .put(req);

        } catch (StatusRuntimeException e) {

            insertExceptions++;

            System.err.println("[INSERT GRPC ERROR] key=" + key +
                    " status=" + e.getStatus());

            return Status.ERROR;

        } catch (Exception e) {

            insertExceptions++;

            System.err.println("[INSERT EXCEPTION] key=" + key);
            e.printStackTrace();

            return Status.ERROR;
        }

        if (!reply.getSuccess()) {
            insertFailures++;
            System.err.println("[INSERT FAILED] key=" + key);
            return Status.ERROR;
        }

        return Status.OK;
    }

    @Override
    public Status update(String table,
                         String key,
                         Map<String, ByteIterator> values) {
        return insert(table, key, values);
    }

    @Override
    public Status delete(String table, String key) {

        Kv.AgentDeleteRequest req =
                Kv.AgentDeleteRequest.newBuilder()
                        .setKey(key)
                        .build();

        Kv.AgentDeleteReply reply;

        try {
            reply = stub
                    .withDeadlineAfter(RPC_DEADLINE_SECONDS, TimeUnit.SECONDS)
                    .delete(req);

        } catch (StatusRuntimeException e) {

            deleteExceptions++;

            System.err.println("[DELETE GRPC ERROR] key=" + key +
                    " status=" + e.getStatus());

            return Status.ERROR;

        } catch (Exception e) {

            deleteExceptions++;

            System.err.println("[DELETE EXCEPTION] key=" + key);
            e.printStackTrace();

            return Status.ERROR;
        }

        if (!reply.getSuccess()) {
            System.err.println("[DELETE FAILED] key=" + key);
            return Status.ERROR;
        }

        return Status.OK;
    }

    @Override
    public Status scan(String table,
                       String startkey,
                       int recordcount,
                       Set<String> fields,
                       Vector<HashMap<String, ByteIterator>> result) {

        return Status.NOT_IMPLEMENTED;
    }
}
